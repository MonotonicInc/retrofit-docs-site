{{- /* hugo-book's prev-next.html walks the file tree. Here previous and next follow manifest.json's order, the same order as the left navigation (_partials/nav.html), across section boundaries. */ -}} {{- $order := slice -}} {{- range hugo.Data.manifest.sections -}} {{- range .pages -}} {{- with site.GetPage (printf "/%s" (strings.TrimSuffix ".md" .path)) -}} {{- $order = $order | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- $prev := false -}} {{- $next := false -}} {{- range $i, $p := $order -}} {{- if eq $p.RelPermalink $.RelPermalink -}} {{- if gt $i 0 }}{{ $prev = index $order (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $order) }}{{ $next = index $order (add $i 1) }}{{ end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{ with $prev }} {{ partial "docs/title" . }} {{ end }} {{ with $next }} {{ partial "docs/title" . }} {{ end }}
{{- end -}}
